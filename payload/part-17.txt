function parseCsv(text){
  const rows=[]; let row=[], cell='', q=false;
  for(let i=0;i<text.length;i++){
    const c=text[i], next=text[i+1];
    if(c==='"' && q && next==='"'){cell+='"'; i++; continue;}
    if(c==='"'){q=!q; continue;}
    if(c===',' && !q){row.push(cell); cell=''; continue;}
    if((c==='\n'||c==='\r') && !q){ if(c==='\r'&&next==='\n') i++; row.push(cell); cell=''; if(row.some(x=>x.trim()!=='')) rows.push(row); row=[]; continue;}
    cell+=c;
  }
  if(cell!==''||row.length){row.push(cell); if(row.some(x=>x.trim()!=='')) rows.push(row);}
  if(!rows.length) return [];
  const headers=rows.shift().map(h=>h.trim());
  return rows.map(r=>Object.fromEntries(headers.map((h,i)=>[h,(r[i]||'').trim()])));
}

function computeDownstreamPath(set,startId){
  // 平行表示では選択ノードから最も新しい年代へ到達する経路だけを強調する。
  const reachable=new Set([startId]), outEdges=[], stack=[startId]; let guard=0;
  while(stack.length && guard++<20000){
    const id=stack.pop(); const outs=set.adjacency.get(id)||[];
    for(const e of outs){
      const t=e.target; if(!set.map.has(t)) continue;
      outEdges.push(e);
      if(!reachable.has(t)){ reachable.add(t); stack.push(t); }
    }
  }
  let latest=-Infinity;
  for(const id of reachable){ const n=set.map.get(id); if(n) latest=Math.max(latest,nodeYear(n)); }
  const latestIds=[...reachable].filter(id=>nodeYear(set.map.get(id))===latest);
  const reverse=new Map();
  for(const e of outEdges){ if(!reachable.has(e.source)||!reachable.has(e.target)) continue; if(!reverse.has(e.target)) reverse.set(e.target,[]); reverse.get(e.target).push(e); }
  const ids=new Set(latestIds), edgeKeys=new Set(), back=[...latestIds]; guard=0;
  while(back.length && guard++<20000){
    const id=back.pop();
    if(id===startId) continue;
    for(const e of reverse.get(id)||[]){
      edgeKeys.add(edgeKey(e));
      if(!ids.has(e.source)){ ids.add(e.source); back.push(e.source); }
    }
  }
  ids.add(startId);
  return {ids,edgeKeys,latestYear:latest,latestIds,mode:'future'};
}
function computeTreePastFuturePath(set,startId){
  // ツリー表示では、選択ノードの未来だけでなく過去(祖先)も含めて強調する。
  const descendantReach=new Set([startId]), descendantEdges=[], fwd=[startId]; let guard=0;
  while(fwd.length && guard++<40000){
    const id=fwd.pop();
    for(const e of (set.adjacency.get(id)||[])){
      if(!set.map.has(e.target)) continue;
      descendantEdges.push(e);
      if(!descendantReach.has(e.target)){ descendantReach.add(e.target); fwd.push(e.target); }
    }
  }
  let latest=-Infinity;
  for(const id of descendantReach){ const n=set.map.get(id); if(n) latest=Math.max(latest,nodeYear(n)); }
  const latestIds=[...descendantReach].filter(id=>nodeYear(set.map.get(id))===latest);

  const ancestorIds=new Set([startId]), ancestorEdgeKeys=new Set(), back=[startId]; guard=0;
  while(back.length && guard++<40000){
    const id=back.pop();
    for(const e of (set.reverseAdjacency.get(id)||[])){
      if(!set.map.has(e.source)) continue;
      ancestorEdgeKeys.add(edgeKey(e));
      if(!ancestorIds.has(e.source)){ ancestorIds.add(e.source); back.push(e.source); }
    }
  }

  const downReverse=new Map();
  for(const e of descendantEdges){
    if(!downReverse.has(e.target)) downReverse.set(e.target,[]);
    downReverse.get(e.target).push(e);
  }
  const descendantIds=new Set(latestIds), descendantEdgeKeys=new Set(), downBack=[...latestIds]; guard=0;
  while(downBack.length && guard++<40000){
    const id=downBack.pop();
    if(id===startId) continue;

      descendantEdgeKeys.add(edgeKey(e));
      if(!descendantIds.has(e.source)){ descendantIds.add(e.source); downBack.push(e.source); }
    }
  }
  descendantIds.add(startId);

  const ids=new Set([...ancestorIds, ...descendantIds]);
  const edgeKeys=new Set([...ancestorEdgeKeys, ...descendantEdgeKeys]);
  return {ids,edgeKeys,latestYear:latest,latestIds,mode:'past-future'};
}

function safeSourceUrl(url){
  const raw=String(url??'').trim();
  if(!raw) return '';
  try{
    const u=new URL(raw,location.href);
    if(u.protocol!=='https:'&&u.protocol!=='http:') return '';
    // 空URLや誤ってアプリ自身を指すURLは「出典」として表示しない。
    const here=new URL(location.href);
    const norm=p=>p.replace(/\/+$/,'/')||'/';
    if(u.origin===here.origin && norm(u.pathname)===norm(here.pathname)) return '';
    return u.href;
  }catch(_){ return ''; }
}
function linkHtml(url,label='開く'){
  const u=safeSourceUrl(url);
  return u?`<a href="${esc(u)}" target="_blank" rel="noopener noreferrer">${esc(label)}</a>`:'';
}

  for(const [key,links] of Object.entries(EVENT_SOURCE_LINKS)){

function collectNodeSourceLinks(n,mo){
  const out=[],seen=new Set();

    if(!u||seen.has(u)) return;
    seen.add(u); out.push({label:label||u,url:u,note});
  };
  // ノード固有URLがある場合は最優先。

  push(mo&&mo.source||'指標出典',mo&&mo.source_url||'','選択中の規模指標');

  const finance=(GRAPH.meta&&GRAPH.meta.financial_metric_sources)||{};

  const names=[raw,...raw.split(/[／/]/)].map(s=>s.replace(/等$/,'').trim()).filter(Boolean);
  for(const name of [...new Set(names)]){
    const s=finance[name];
    if(s&&s.url) push(s.source||name,s.url,'現在/後身銀行の会社概要・指標参照元');
  }
  // 出来事を直接裏付ける一次資料・公式沿革・報道を優先して追加。
  appendContextualSourceLinks(n,push);
  // 歴史変遷の共通原典。
  for(const s of CORE_SOURCE_LINKS) push(s.label,s.url,s.note);
  if(n.no===1 || n.no_min===1 || /第一国立銀行/.test(String(n.label||''))){

  }
  return out;
}

function applyFilters(){
  const set=activeSet();
  const showBranches=document.getElementById('toggleBranches').checked;
  const q=searchTerm;
  let matches=0;
  const visibleIds=new Set();

  for(const obj of set.nodeObjs){
    const n=obj.userData.node;
    let vis=(showBranches||!n.branch);
    if(q){
      vis=vis&&qMatchesNode(n);
      if(vis) matches++;
    }
    obj.visible=vis;
    if(vis) visibleIds.add(n.id);
    obj.scale.setScalar(q&&vis?1.8:1);
    obj.material.opacity=q&&vis?1:(n.branch?.84:.98);
  }

  for(const bar of set.barObjs){
    const n=bar.userData.node;
    bar.visible=bars && (showBranches||!n.branch) && (!q||visibleIds.has(n.id));
  }

  for(const spr of set.labelObjs){
    const n=spr.userData.node;
    spr.visible=document.getElementById('toggleLabels').checked &&
      (showBranches||!n.branch) &&
      (!q||visibleIds.has(n.id));
  }

  for(const obj of set.edgeObjs){
    const br=!!obj.userData.branch;
    const e=obj.userData.edge||{};
    const sourceVisible=visibleIds.has(e.source);
    const targetVisible=visibleIds.has(e.target);
    obj.visible=(showBranches||!br) && (!q || (sourceVisible && targetVisible));
  }

  updateLabelVisibility();
  updateStats(q?matches:null);
}
