function parseCsv(text){
  const rows=[]; let row=[], cell='', q=false;
  for(let i=0;i<text.length;i++){
    const c=text[i], next=text[i+1];
    if(c==='"' && q && next==='"'){cell+='"'; i++; continue;}
    if(c==='"'){q=!q; continue;}
    if(c===',' && !q){row.push(cell); cell=''; continue;}
    if((c==='\n'||c==='\r') && !q){ if(c==='\r'&&next==='\n') i++; row.push(cell); cell=''; if(row.some(x=>x.trim()!=='')) rows.push(row); row=[]; continue;}
    cell+=c;
  }
  if(cell!==''||row.length){row.push(cell); if(row.some(x=>x.trim()!=='')) rows.push(row);}
  if(!rows.length) return [];
  const headers=rows.shift().map(h=>h.trim());
  return rows.map(r=>Object.fromEntries(headers.map((h,i)=>[h,(r[i]||'').trim()])));
}

function computeDownstreamPath(set,startId){
  // 平行表示では選択ノードから最も新しい年代へ到達する経路だけを強調する。
  const reachable=new Set([startId]), outEdges=[], stack=[startId]; let guard=0;
  while(stack.length && guard++<20000){
    const id=stack.pop(); const outs=set.adjacency.get(id)||[];
    for(const e of outs){
      const t=e.target; if(!set.map.has(t)) continue;
      outEdges.push(e);
      if(!reachable.has(t)){ reachable.add(t); stack.push(t); }
    }
  }
  let latest=-Infinity;
  for(const id of reachable){ const n=set.map.get(id); if(n) latest=Math.max(latest,nodeYear(n)); }
  const latestIds=[...reachable].filter(id=>nodeYear(set.map.get(id))===latest);
  const reverse=new Map();
  for(const e of outEdges){ if(!reachable.has(e.source)||!reachable.has(e.target)) continue; if(!reverse.has(e.target)) reverse.set(e.target,[]); reverse.get(e.target).push(e); }
  const ids=new Set(latestIds), edgeKeys=new Set(), back=[...latestIds]; guard=0;
  while(back.length && guard++<20000){
    const id=back.pop();
    if(id===startId) continue;
    for(const e of reverse.get(id)||[]){
      edgeKeys.add(edgeKey(e));
      if(!ids.has(e.source)){ ids.add(e.source); back.push(e.source); }
    }
  }
  ids.add(startId);
  return {ids,edgeKeys,latestYear:latest,latestIds,mode:'future'};
}
function computeTreePastFuturePath(set,startId){
  // ツリー表示では、選択ノードの未来だけでなく過去(祖先)も含めて強調する。
  const descendantReach=new Set([startId]), descendantEdges=[], fwd=[startId]; let guard=0;
  while(fwd.length && guard++<40000){
    const id=fwd.pop();
    for(const e of (set.adjacency.get(id)||[])){
      if(!set.map.has(e.target)) continue;
      descendantEdges.push(e);
      if(!descendantReach.has(e.target)){ descendantReach.add(e.target); fwd.push(e.target); }
    }
  }
  let latest=-Infinity;
  for(const id of descendantReach){ const n=set.map.get(id); if(n) latest=Math.max(latest,nodeYear(n)); }
  const latestIds=[...descendantReach].filter(id=>nodeYear(set.map.get(id))===latest);

  const ancestorIds=new Set([startId]), ancestorEdgeKeys=new Set(), back=[startId]; guard=0;
  while(back.length && guard++<40000){
    const id=back.pop();
    for(const e of (set.reverseAdjacency.get(id)||[])){
      if(!set.map.has(e.source)) continue;
      ancestorEdgeKeys.add(edgeKey(e));
      if(!ancestorIds.has(e.source)){ ancestorIds.add(e.source); back.push(e.source); }
    }
  }

  const downReverse=new Map();
  for(const e of descendantEdges){
    if(!downReverse.has(e.target)) downReverse.set(e.target,[]);
    downReverse.get(e.target).push(e);
  }
  const descendantIds=new Set(latestIds), descendantEdgeKeys=new Set(), downBack=[...latestIds]; guard=0;
  while(downBack.length && guard++<40000){
    const id=downBack.pop();
    if(id===startId) continue;

      descendantEdgeKeys.add(edgeKey(e));
      if(!descendantIds.has(e.source)){ descendantIds.add(e.source); downBack.push(e.source); }
    }
  }
  descendantIds.add(startId);

  const ids=new Set([...ancestorIds, ...descendantIds]);
  const edgeKeys=new Set([...ancestorEdgeKeys, ...descendantEdgeKeys]);
  return {ids,edgeKeys,latestYear:latest,latestIds,mode:'past-future'};
}

  for(const n of nodes){ n.x=Number(n.x ?? yearToX(n.display_year||n.year||YEAR_MIN)); n.y=Number(n.y ?? noToY(n.no||0)); n.z=Number(n.z||0); n._baseX=n.x; n._baseY=n.y; n._baseZ=n.z; map.set(n.id,n); }
  for(const e of edges){ if(map.has(e.source) && map.has(e.target)){ if(!adjacency.has(e.source)) adjacency.set(e.source,[]); adjacency.get(e.source).push(e); if(!reverseAdjacency.has(e.target)) reverseAdjacency.set(e.target,[]); reverseAdjacency.get(e.target).push(e); } }
  for(const e of edges){ const a=map.get(e.source),b=map.get(e.target); if(!a||!b) continue; const c=edgeColor(e); const baseOpacity=e.type==='国立銀行間合流'?.24:.58; const mat=new THREE.LineBasicMaterial({color:c,transparent:true,opacity:baseOpacity}); const line=new THREE.Line(new THREE.BufferGeometry().setFromPoints(edgePoints(a,b)),mat); line.userData={edge:e,edgeKey:edgeKey(e),branch:e.type!=='主系譜',baseColor:c,baseOpacity}; group.add(line); edgeObjs.push(line); const dir=new THREE.Vector3(b.x-a.x,b.y-a.y,(b.z||0)-(a.z||0)); const len=dir.length(); if(len>1){dir.normalize(); const cone=new THREE.Mesh(new THREE.ConeGeometry(e.type==='主系譜'?1.7:2.1,e.type==='主系譜'?4.8:6.0,10),new THREE.MeshBasicMaterial({color:c,transparent:true,opacity:e.type==='国立銀行間合流'?.22:.62})); cone.position.set(b.x-dir.x*4,b.y-dir.y*4,(b.z||0)-dir.z*4); cone.quaternion.setFromUnitVectors(new THREE.Vector3(0,1,0),dir); cone.userData={edge:e,edgeKey:edgeKey(e),branch:e.type!=='主系譜',baseColor:c,baseOpacity:e.type==='国立銀行間合流'?.22:.62}; group.add(cone); edgeObjs.push(cone);} }
  const sphere=new THREE.SphereGeometry(kind==='tree'?3.5:2.8,18,12), box=new THREE.BoxGeometry(5.4,5.4,5.4);
  for(const n of nodes){ const color=nodeColor(n); const mat=new THREE.MeshPhongMaterial({color,emissive:color,emissiveIntensity:.07,transparent:true,opacity:n.branch?.86:1}); const geom=n.type==='商号変更'?box:sphere; const mesh=new THREE.Mesh(geom,mat); mesh.position.set(n.x,n.y,n.z||0); mesh.userData={node:n,branch:!!n.branch,kind,baseColor:color,baseOpacity:n.branch?.86:1,baseEmissiveIntensity:.07}; group.add(mesh); nodeObjs.push(mesh); nodeMeshMap.set(n.id,mesh);
    const bh=metricHeight(n,activeMetric); const missing=metricValue(n,activeMetric)==null; const barGeom=new THREE.BoxGeometry(2.6,2.6,1); const barMat=new THREE.MeshPhongMaterial({color,emissive:color,emissiveIntensity:.04,transparent:true,opacity:missing?.18:.62}); const bar=new THREE.Mesh(barGeom,barMat); bar.scale.z=Math.max(.5,bh); bar.position.set(n.x,n.y,bh/2); bar.userData={bar:true,node:n,branch:!!n.branch,metricHeight:bh,baseColor:color,baseOpacity:missing?.18:.62}; group.add(bar); barObjs.push(bar); barMap.set(n.id,bar);
    const lab=(n.type==='国立銀行' && kind==='parallel')?`${n.no}. ${n.label}`:(n.label||''); const tt=makeTextTexture(lab,n.type==='商号変更'?'#fff0b8':'#f3f4f6',n.branch?'rgba(24,22,31,.84)':'rgba(7,9,12,.80)'); const spr=new THREE.Sprite(new THREE.SpriteMaterial({map:tt.tex,transparent:true,opacity:n.branch?.85:.92,depthWrite:false})); spr.position.set(n.x+7,n.y+2,(n.z||0)+4); spr.scale.set(tt.w/12,tt.h/12,1); spr.userData={node:n,branch:!!n.branch,isLabel:true,kind,labelText:lab,labelPxW:tt.w,labelPxH:tt.h,baseOpacity:n.branch?.85:.92,baseScaleX:spr.scale.x,baseScaleY:spr.scale.y}; group.add(spr); labelObjs.push(spr); labelMap.set(n.id,spr);
  }
  return {group,map,adjacency,reverseAdjacency,edges,nodeObjs,labelObjs,edgeObjs,barObjs,nodeMeshMap,labelMap,barMap};
}
function parallelNodeNo(n){
  const v=Number(n&&(n.no??n.no_min??n.no_max));
  return Number.isFinite(v)?v:null;
}
function parallelRowY(no){
  const y=noToY(no);
  return parallelBranchExpansion && Number(no)>parallelBranchExpansion.hostNo ? y-parallelBranchExpansion.extraHeight : y;
}
function syncParallelNodePosition(n){
  const set=groups.parallel;if(!set||!n)return;
  const mesh=set.nodeMeshMap.get(n.id),bar=set.barMap.get(n.id),spr=set.labelMap.get(n.id);
  if(mesh)mesh.position.set(n.x,n.y,n.z||0);
  if(bar)bar.position.set(n.x,n.y,(bar.userData.metricHeight||.5)/2);
  if(spr)spr.position.set(n.x+7,n.y+2,(n.z||0)+4);
}
function refreshParallelEdgeGeometry(){
  const set=groups.parallel;if(!set)return;
  for(const obj of set.edgeObjs){
    const e=obj.userData&&obj.userData.edge;if(!e)continue;
    const a=set.map.get(e.source),b=set.map.get(e.target);if(!a||!b)continue;
    if(obj.type==='Line'){
      obj.geometry.setFromPoints(edgePoints(a,b));
      if(obj.geometry.computeBoundingSphere)obj.geometry.computeBoundingSphere();
    }else{
      const dir=new THREE.Vector3(b.x-a.x,b.y-a.y,(b.z||0)-(a.z||0)),len=dir.length();
      if(len>.1){dir.normalize();obj.position.set(b.x-dir.x*4,b.y-dir.y*4,(b.z||0)-dir.z*4);obj.quaternion.setFromUnitVectors(new THREE.Vector3(0,1,0),dir);}
    }
  }
}
function refreshParallelGridGeometry(){
  const g=scene&&scene.getObjectByName('grid');if(!g)return;
  for(const line of g.children){
    if(line.userData&&line.userData.rowNo){
      const y=parallelRowY(line.userData.rowNo);
      line.geometry.setFromPoints([new THREE.Vector3(yearToX(YEAR_MIN),y,0),new THREE.Vector3(yearToX(YEAR_MAX),y,0)]);
    }else if(line.userData&&line.userData.gridYear){
      const x=yearToX(line.userData.gridYear);
      line.geometry.setFromPoints([new THREE.Vector3(x,parallelRowY(1),0),new THREE.Vector3(x,parallelRowY(153),0)]);
    }
    if(line.geometry&&line.geometry.computeBoundingSphere)line.geometry.computeBoundingSphere();
  }
}
function resetParallelBranchExpansion(){
  const set=groups.parallel;
  parallelBranchExpansion=null;
  if(!set)return;
  for(const n of set.map.values()){
    n.x=Number(n._baseX??n.x);n.y=Number(n._baseY??n.y);n.z=Number(n._baseZ??n.z);
    syncParallelNodePosition(n);
  }
  refreshParallelEdgeGeometry();
  refreshParallelGridGeometry();
}
function branchComponentFromSeed(set,seedId){
  const ids=new Set(),stack=[seedId];let guard=0;
  while(stack.length&&guard++<10000){
    const id=stack.pop();if(ids.has(id))continue;
    const n=set.map.get(id);if(!n||!n.branch)continue;
    ids.add(id);
    for(const e of (set.adjacency.get(id)||[])){const t=set.map.get(e.target);if(t&&t.branch&&!ids.has(t.id))stack.push(t.id);}
    for(const e of (set.reverseAdjacency.get(id)||[])){const a=set.map.get(e.source);if(a&&a.branch&&!ids.has(a.id))stack.push(a.id);}
  }
  return ids;
}
function findParallelBranchLanes(startId){
  const set=groups.parallel,start=set&&set.map.get(startId);if(!set||!start)return null;
  const seeds=[];
  if(start.branch)seeds.push(start.id);
  for(const e of (set.reverseAdjacency.get(start.id)||[])){const a=set.map.get(e.source);if(a&&a.branch)seeds.push(a.id);}
  for(const e of (set.adjacency.get(start.id)||[])){const b=set.map.get(e.target);if(b&&b.branch)seeds.push(b.id);}
  if(!seeds.length)return null;
  const comps=[],seen=new Set();
  for(const seed of seeds){
    if(seen.has(seed))continue;
    const ids=branchComponentFromSeed(set,seed);if(!ids.size)continue;
    for(const id of ids)seen.add(id);
    const nodes=[...ids].map(id=>set.map.get(id)).filter(Boolean).sort((a,b)=>nodeYear(a)-nodeYear(b)||(a._baseZ||0)-(b._baseZ||0));
    const first=nodes.find(n=>n.type==='合併相手開始')||nodes[0];
    const label=String(first&&first.label||'支流').replace(/（開始年未精査）|（開始）|（第\d+系）/g,'');
    const z=Math.min(...nodes.map(n=>Number(n._baseZ||0)));
    comps.push({ids,label,z,firstYear:nodeYear(first)});
  }
  if(!comps.length)return null;
  comps.sort((a,b)=>a.z-b.z||a.firstYear-b.firstYear||a.label.localeCompare(b.label,'ja'));
  const hostNo=parallelNodeNo(start)??parallelNodeNo(set.map.get(seeds[0]));
  return Number.isFinite(hostNo)?{hostNo,lanes:comps}:null;
}
function setParallelBranchExpansion(startId){
  resetParallelBranchExpansion();
  const spec=findParallelBranchLanes(startId);if(!spec)return;
  const set=groups.parallel,laneGap=Math.max(7.6,Y_GAP*.82),extraHeight=laneGap*spec.lanes.length+Math.max(1.8,Y_GAP*.18);
  const revealedIds=new Set();
  parallelBranchExpansion={selectedId:startId,hostNo:spec.hostNo,laneGap,extraHeight,lanes:[],revealedIds};
  for(const n of set.map.values()){
    const no=parallelNodeNo(n);
    n.y=Number(n._baseY??n.y)-(no!==null&&no>spec.hostNo?extraHeight:0);
    n.z=Number(n._baseZ??n.z);
  }
  spec.lanes.forEach((lane,i)=>{
    const y=noToY(spec.hostNo)-laneGap*(i+1);
    const info={label:lane.label,y,ids:lane.ids};
    parallelBranchExpansion.lanes.push(info);
    for(const id of lane.ids){
      revealedIds.add(id);
      const n=set.map.get(id);if(!n)continue;
      n.y=y;n.z=0;
    }
  });
  for(const n of set.map.values())syncParallelNodePosition(n);
  refreshParallelEdgeGeometry();
  refreshParallelGridGeometry();
}

function hoverBaseScale(id,mesh){

function restoreHoverAccent(){
  if(!hoverAccentId) return;
  const set=activeSet(), mesh=set&&set.nodeMeshMap&&set.nodeMeshMap.get(hoverAccentId);
  if(mesh){
    const inPath=hoverPathIds&&hoverPathIds.has(hoverAccentId);
    mesh.scale.setScalar(hoverBaseScale(hoverAccentId,mesh));
    const base=mesh.userData.baseColor||0xb6c5dd,hi=highlightColors();
    mesh.material.color.setHex(inPath?hi.node:base);
    mesh.material.emissive.setHex(inPath?hi.emissive:base);
    mesh.material.emissiveIntensity=inPath?.95:(mesh.userData.baseEmissiveIntensity||.07);
  }
  hoverAccentId=null;
}
function setHoverAccent(id){
  if(hoverAccentId===id) return;
  restoreHoverAccent();
  hoverAccentId=id;
}
function updateHover(e){
  const h=pick(e),tip=document.getElementById('tooltip');
  // 情報はクリックしたノードのパネルに固定し、ホバーでは内容を切り替えない。
  tip.style.display='none';
  if(h&&h.object.userData.node){
    const n=h.object.userData.node;
    if(lockedNodeId){
      // クリック固定中も、別ノードにカーソルを置いたときの強調アニメーションだけは維持する。
      setHoverAccent(n.id);
    }else{
      restoreHoverAccent();
      setPathHighlight(n.id,false);
    }
  }else{
    if(lockedNodeId) restoreHoverAccent();
    else clearPathHighlight();
  }
}

function safeSourceUrl(url){
  const raw=String(url??'').trim();
  if(!raw) return '';
  try{
    const u=new URL(raw,location.href);
    if(u.protocol!=='https:'&&u.protocol!=='http:') return '';
    // 空URLや誤ってアプリ自身を指すURLは「出典」として表示しない。
    const here=new URL(location.href);
    const norm=p=>p.replace(/\/+$/,'/')||'/';
    if(u.origin===here.origin && norm(u.pathname)===norm(here.pathname)) return '';
    return u.href;
  }catch(_){ return ''; }
}
function linkHtml(url,label='開く'){
  const u=safeSourceUrl(url);
  return u?`<a href="${esc(u)}" target="_blank" rel="noopener noreferrer">${esc(label)}</a>`:'';
}

  const finance=(GRAPH.meta&&GRAPH.meta.financial_metric_sources)||{};

    if(s&&s.url) push(s.source||name,s.url,'現在/後身銀行の会社概要・指標参照元');
  }
  // 出来事を直接裏付ける一次資料・公式沿革・報道を優先して追加。
  appendContextualSourceLinks(n,push);
  // 「背景・理由」の根拠となる研究論文・公的資料・プレスリリース。
  appendReasonSourceLinks(n,push);
  // 歴史変遷の共通原典。
  for(const s of CORE_SOURCE_LINKS) push(s.label,s.url,s.note);
  if(n.no===1 || n.no_min===1 || /第一国立銀行/.test(String(n.label||''))){
    push('日本銀行「日本で最初にできた銀行はどこですか？」','https://www.boj.or.jp/about/education/oshiete/history/j10.htm','第一国立銀行に関する日本銀行の解説');
  }
  return out;
}

function applyFilters(){
  const set=activeSet();
  const showBranches=document.getElementById('toggleBranches').checked;
  const q=searchTerm;
  let matches=0;
  const visibleIds=new Set();

  for(const obj of set.nodeObjs){
    const n=obj.userData.node;
    let vis=(showBranches||!n.branch);
    if(q){
      vis=vis&&qMatchesNode(n);
      if(vis) matches++;
    }
    obj.visible=vis;
    if(vis) visibleIds.add(n.id);
    obj.scale.setScalar(q&&vis?1.8:1);
    obj.material.opacity=q&&vis?1:(n.branch?.84:.98);
  }

  for(const bar of set.barObjs){
    const n=bar.userData.node;
    bar.visible=bars && (showBranches||!n.branch) && (!q||visibleIds.has(n.id));
  }

  for(const spr of set.labelObjs){
    const n=spr.userData.node;
    spr.visible=document.getElementById('toggleLabels').checked &&
      (showBranches||!n.branch) &&
      (!q||visibleIds.has(n.id));
  }

  for(const obj of set.edgeObjs){
    const br=!!obj.userData.branch;
    const e=obj.userData.edge||{};
    const sourceVisible=visibleIds.has(e.source);
    const targetVisible=visibleIds.has(e.target);
    obj.visible=(showBranches||!br) && (!q || (sourceVisible && targetVisible));
  }

  updateLabelVisibility();
  updateStats(q?matches:null);
}

function worldXToFixedScreenX(x){
  // 年号軸は3D投影に従わせず、横方向のパン/ズームだけに連動させる。
